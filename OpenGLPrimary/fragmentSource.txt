out vec4 color;

in vec2 texCoord;

uniform sampler2D outTexture;
uniform sampler2D outFace;
uniform vec4 outColor;	
uniform float value;
void main() {													
     color =mix( texture(outTexture,texCoord) , texture(outFace,texCoord),value);
}
